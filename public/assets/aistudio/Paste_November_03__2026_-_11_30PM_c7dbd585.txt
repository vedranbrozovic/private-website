    // Standard NBA schedule matrix head-to-head games per pair of teams
    function getPairSeasonGames(teamA, teamB) {
      if (teamA.div === teamB.div) return 4;
      if (teamA.conf === teamB.conf) return 3.6;
      return 2;
    }

    // Generate full 30-pick Snake Order array of player indices (e.g., 0,1,2, 2,1,0, 0,1,2...)
    function generateSnakeOrder(numPlayers) {
      const order = [];
      const rounds = 30 / numPlayers;
      for (let r = 0; r < rounds; r++) {
        const roundIndices = Array.from({ length: numPlayers }, (_, i) => i);
        if (r % 2 === 1) roundIndices.reverse();
        order.push(...roundIndices);
      }
      return order;
    }

            for (let j = i + 1; j < ownedTeamObjs.length; j++) {

          const rawTotalSeasonGames = ownedTeamObjs.length * 82;
          const seasonProgressRatio = rawTotalSeasonGames > 0 ? totalGamesPlayed / rawTotalSeasonGames : 0;
          const estimatedInternalGamesPlayedSoFar = Math.round(totalInternalGamesSeason * seasonProgressRatio);
          const remainingInternalGames = Math.max(0, totalInternalGamesSeason - estimatedInternalGamesPlayedSoFar);

          // Possible Points (To Date): Total games played minus guaranteed losses from internal clashes
          const possiblePointsToDate = Math.max(totalWins, totalGamesPlayed - estimatedInternalGamesPlayedSoFar);

          // Remaining Season Ceiling: Current Wins + Remaining Games - Remaining Forced Internal Losses
          const remainingTeamGames = Math.max(0, rawTotalSeasonGames - totalGamesPlayed);
          const maxPossibleSeasonPoints = totalWins + remainingTeamGames - remainingInternalGames;
          const initialSeasonCeiling = rawTotalSeasonGames - totalInternalGamesSeason;

            initialSeasonCeiling,
            totalInternalGamesSeason

                <label className="px-4 py-2.5 rounded-xl bg-slate-800 hover:bg-slate-700 border border-slate-700 text-xs font-bold text-slate-200 cursor-pointer transition">
                  📂 Load Saved League (.json)

                </label>
                <button
                  onClick={startDraft}
                  className="w-full sm:w-auto px-8 py-3.5 rounded-xl bg-gradient-to-r from-orange-500 to-amber-500 hover:from-orange-400 hover:to-amber-400 text-white font-black shadow-lg shadow-orange-500/25 transition"

              </div>
            </div>

            {/* Snake Order Ribbon */}
            <div className="bg-slate-900/90 border border-slate-800 rounded-2xl p-4 mb-6 overflow-x-auto custom-scrollbar">
              <div className="flex items-center gap-2 min-w-max">
                {snakeOrder.map((pIdx, idx) => {
                  const p = league.players[pIdx];
                  const pickRecord = league.picks[idx];
                  const pickedTeam = pickRecord ? NBA_TEAMS.find((t) => t.id === pickRecord.teamId) : null;
                  const isCurrent = idx === currentPickIdx;

                  return (
                    <div
                      key={idx}
                      className={`w-28 p-2.5 rounded-xl border transition shrink-0 ${
                        isCurrent
                          ? 'bg-orange-500/20 border-orange-500 ring-2 ring-orange-500/40'
                          : pickRecord
                          ? 'bg-slate-800/40 border-slate-800 opacity-75'
                          : 'bg-slate-800/80 border-slate-700/70'
                      }`}
                    >
                      <div className="flex items-center justify-between text-[11px] font-bold text-slate-400 mb-1">
                        <span>Pick #{idx + 1}</span>
                        <span>R{Math.floor(idx / league.numPlayers) + 1}</span>
                      </div>
                      <div className="font-bold text-xs truncate text-white">{p.name}</div>
                      {pickedTeam ? (
                        <div className="flex items-center gap-1.5 mt-1.5 pt-1.5 border-t border-slate-700/60">
                          <img src={getLogoUrl(pickedTeam.abbr)} className="w-4 h-4 object-contain" />
                          <span className="text-xs font-extrabold text-orange-300">{pickedTeam.abbr}</span>
                        </div>
                      ) : (
                        <div className="text-[11px] text-slate-500 mt-1.5 pt-1.5 border-t border-slate-800">
                          {isCurrent ? 'Picking...' : 'Waiting'}
                        </div>
                      )}
                    </div>
                  );
                })}
              </div>
            </div>

            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
              {/* Available Teams Grid */}
              <div className="lg:col-span-2 bg-slate-900/90 border border-slate-800 rounded-2xl p-5">
                <div className="flex flex-wrap items-center justify-between gap-3 mb-5">
                  <div className="flex items-center gap-2">
                    {['ALL', 'East', 'West'].map((c) => (
                      <button
                        key={c}
                        onClick={() => setConfFilter(c)}
                        className={`px-3.5 py-1.5 rounded-lg text-xs font-bold transition ${
                          confFilter === c ? 'bg-orange-500 text-white' : 'bg-slate-800 text-slate-400 hover:text-white'
                        }`}
                      >
                        {c === 'ALL' ? 'All Teams' : `${c}ern Conf`}
                      </button>
                    ))}
                  </div>
                  <input
                    type="text"
                    placeholder="Search team..."
                    value={searchQuery}
                    onChange={(e) => setSearchQuery(e.target.value)}
                    className="bg-slate-800 border border-slate-700 rounded-xl px-3.5 py-1.5 text-sm text-white focus:outline-none focus:border-orange-500"
                  />
                </div>

                <div className="grid grid-cols-2 sm:grid-cols-3 gap-3 max-h-[540px] overflow-y-auto pr-1 custom-scrollbar">
                  {availableTeams.map((team) => {
                    const rec = teamRecords[team.id] || { wins: 0, losses: 0 };
                    return (
                      <button
                        key={team.id}
                        onClick={() => makeDraftPick(team.id)}
                        className="group flex flex-col items-start p-3.5 rounded-2xl bg-slate-800/60 hover:bg-slate-800 border border-slate-700/70 hover:border-orange-500 transition text-left"
                      >
                        <div className="flex items-center justify-between w-full mb-2">
                          <img
                            src={getLogoUrl(team.abbr)}
                            alt={team.name}
                            className="w-11 h-11 object-contain group-hover:scale-110 transition"
                          />
                          <span className="text-[11px] font-bold px-2 py-0.5 rounded-full bg-slate-900/80 text-slate-300">
                            {team.conf} • {team.div}
                          </span>
                        </div>
                        <div className="font-extrabold text-sm text-white group-hover:text-orange-400 transition">
                          {team.name}
                        </div>
                        <div className="text-xs text-slate-400 mt-1">
                          Record: <span className="text-slate-200 font-semibold">{rec.wins}-{rec.losses}</span>
                        </div>
                      </button>
                    );
                  })}
                </div>
              </div>

              {/* Drafted Rosters */}
              <div className="bg-slate-900/90 border border-slate-800 rounded-2xl p-5">
                <h2 className="text-base font-black uppercase tracking-wider text-slate-300 mb-4">
                  Drafted Rosters
                </h2>
                <div className="space-y-4">
                  {league.players.map((p, idx) => (
                    <div key={p.id} className="p-3.5 rounded-xl bg-slate-800/50 border border-slate-700/60">
                      <div className="flex items-center justify-between mb-2">
                        <div className="flex items-center gap-2">
                          <span className={`w-3 h-3 rounded-full bg-gradient-to-r ${PLAYER_COLORS[idx]}`} />
                          <span className="font-bold text-sm text-white">{p.name}</span>
                        </div>
                        <span className="text-xs text-slate-400 font-semibold">
                          {p.teams.length} / {30 / league.numPlayers}
                        </span>
                      </div>
                      <div className="flex flex-wrap gap-1.5">
                        {p.teams.map((tId) => {
                          const t = NBA_TEAMS.find((team) => team.id === tId);
                          return (
                            <div
                              key={tId}
                              className="flex items-center gap-1.5 px-2.5 py-1 rounded-lg bg-slate-900 border border-slate-700 text-xs font-bold"
                            >
                              <img src={getLogoUrl(t.abbr)} className="w-4 h-4 object-contain" />
                              <span>{t.abbr}</span>
                            </div>
                          );
                        })}
                        {p.teams.length === 0 && (
                          <span className="text-xs text-slate-500 italic">No teams drafted yet</span>
                        )}
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          </div>
        );
      }

      // =========================================================================
      // VIEW 3: ACTIVE SEASON DASHBOARD
      // =========================================================================
      return (
        <div className="max-w-7xl mx-auto px-4 py-8">
          {/* Header */}
          <div className="flex flex-wrap items-center justify-between gap-4 mb-8 pb-6 border-b border-slate-800">
            <div className="flex items-center gap-4">
              <div className="w-12 h-12 rounded-2xl bg-gradient-to-br from-orange-500 to-amber-600 flex items-center justify-center text-2xl shadow-lg shadow-orange-500/20">
                🏆
              </div>
              <div>
                <div className="flex items-center gap-2">
                  <span className="text-xs font-bold uppercase tracking-widest px-2.5 py-0.5 rounded-full bg-orange-500/20 text-orange-400 border border-orange-500/30">
                    {league.seasonYear === '2027' ? '2026–27 Season' : '2025–26 Season'}
                  </span>
                  {lastUpdated && (
                    <span className="text-xs text-slate-400">Updated {lastUpdated}</span>
                  )}
                </div>
                <h1 className="text-2xl sm:text-3xl font-black text-white mt-0.5">
                  NBA Wins Pool Standings
                </h1>
              </div>
            </div>

            <div className="flex flex-wrap items-center gap-2.5">
              <select
                value={league.seasonYear}
                onChange={(e) => setLeague({ ...league, seasonYear: e.target.value })}
                className="bg-slate-800 border border-slate-700 rounded-xl px-3 py-2 text-xs font-bold text-slate-200"
              >
                <option value="2027">2026–27 Season</option>
                <option value="2026">2025–26 Season</option>
              </select>

              <button
                onClick={() => fetchNbaData(league.seasonYear)}
                disabled={loadingStats}
                className="px-3.5 py-2 rounded-xl bg-slate-800 hover:bg-slate-700 border border-slate-700 text-xs font-bold text-white transition"
              >
                {loadingStats ? 'Refreshing...' : '↻ Refresh Scores'}
              </button>

              <button
                onClick={() => setShowDraftRecap(true)}
                className="px-3.5 py-2 rounded-xl bg-slate-800 hover:bg-slate-700 border border-slate-700 text-xs font-bold text-slate-200 transition"
              >
                📋 Draft Board
              </button>

              <button
                onClick={exportLeagueJson}
                className="px-3.5 py-2 rounded-xl bg-emerald-950/70 hover:bg-emerald-900/80 border border-emerald-700/50 text-xs font-bold text-emerald-300 transition"
              >
                ⬇ Save JSON
              </button>

              <label className="px-3.5 py-2 rounded-xl bg-slate-800 hover:bg-slate-700 border border-slate-700 text-xs font-bold text-slate-200 cursor-pointer transition">
                ⬆ Load JSON
                <input type="file" accept=".json" onChange={importLeagueJson} className="hidden" />
              </label>

              <button
                onClick={resetLeague}
                className="px-3 py-2 rounded-xl bg-slate-900 hover:bg-rose-950/60 border border-slate-800 hover:border-rose-800 text-xs font-semibold text-slate-400 hover:text-rose-300 transition"
              >
                Reset
              </button>
            </div>
          </div>

          {/* SCOREBOARD CARDS */}
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-5 mb-10">
            {scoreboard.map((player, rankIdx) => (
              <div
                key={player.id}
                className="bg-slate-900/90 border border-slate-800 rounded-3xl p-6 flex flex-col justify-between shadow-xl relative overflow-hidden"
              >
                <div className={`absolute top-0 left-0 right-0 h-1.5 bg-gradient-to-r ${player.colorClass}`} />

                <div>
                  <div className="flex items-center justify-between mb-4">
                    <div className="flex items-center gap-3">
                      <span className={`w-8 h-8 rounded-xl bg-gradient-to-br ${player.colorClass} flex items-center justify-center font-black text-sm text-white`}>
                        #{rankIdx + 1}
                      </span>
                      <div>
                        <h2 className="text-lg font-black text-white">{player.name}</h2>
                        <span className="text-xs text-slate-400">{player.ownedTeamObjs.length} NBA Franchises</span>
                      </div>
                    </div>
                    <div className="text-right">
                      <div className="text-3xl font-black text-white">{player.totalWins}</div>
                      <div className="text-[11px] font-bold uppercase tracking-wider text-orange-400">Points (Wins)</div>
                    </div>
                  </div>

                  {/* Core Tracking Metrics */}
                  <div className="grid grid-cols-2 gap-3 mb-5">
                    {/* Wins / Total Games Played */}
                    <div className="bg-slate-800/60 border border-slate-700/60 rounded-2xl p-3">
                      <div className="text-[11px] font-bold uppercase text-slate-400">Wins / Games Played</div>
                      <div className="text-lg font-black text-white mt-0.5">
                        {player.totalWins} <span className="text-slate-400 font-semibold">/ {player.totalGamesPlayed}</span>
                      </div>
                      <div className="text-xs text-slate-300 font-semibold mt-0.5">
                        {player.winPct}% Win Rate
                      </div>
                    </div>

                    {/* Points Gained / Total Possible Points */}
                    <div className="bg-slate-800/60 border border-slate-700/60 rounded-2xl p-3">
                      <div className="text-[11px] font-bold uppercase text-slate-400" title="Accounts for guaranteed losses when two of your own teams play each other">
                        Pts / Possible Pts
                      </div>
                      <div className="text-lg font-black text-emerald-400 mt-0.5">
                        {player.totalWins} <span className="text-slate-400 font-semibold">/ {player.possiblePointsToDate}</span>
                      </div>
                      <div className="text-xs text-slate-300 font-semibold mt-0.5">
                        Max Season Ceiling: <span className="text-white">{player.maxPossibleSeasonPoints}</span>
                      </div>
                    </div>
                  </div>

                  {/* Owned Franchises List */}
                  <div className="space-y-2">
                    <div className="text-xs font-bold uppercase tracking-wider text-slate-400 mb-2">
                      Drafted Teams
                    </div>
                    <div className="grid grid-cols-2 gap-2 max-h-60 overflow-y-auto pr-1 custom-scrollbar">
                      {player.ownedTeamObjs.map((t) => {
                        const rec = teamRecords[t.id] || { wins: 0, losses: 0 };
                        return (
                          <div
                            key={t.id}
                            className="flex items-center justify-between p-2 rounded-xl bg-slate-800/40 border border-slate-800"
                          >
                            <div className="flex items-center gap-2 min-w-0">
                              <img src={getLogoUrl(t.abbr)} alt={t.abbr} className="w-6 h-6 object-contain shrink-0" />
                              <div className="truncate">
                                <div className="text-xs font-bold text-white leading-tight">{t.abbr}</div>
                                <div className="text-[10px] text-slate-400">{rec.wins}-{rec.losses}</div>
                              </div>
                            </div>
                            <span className="text-xs font-extrabold text-orange-400 px-1.5 py-0.5 rounded bg-orange-500/10">
                              +{rec.wins}
                            </span>
                          </div>
                        );
                      })}
                    </div>
                  </div>
                </div>

                <div className="mt-4 pt-3 border-t border-slate-800/80 flex items-center justify-between text-[11px] text-slate-400">
                  <span>Own-Team Clashes: ~{player.totalInternalGamesSeason}/yr</span>
                  <span>Pre-Season Max: {player.initialSeasonCeiling} pts</span>
                </div>
              </div>
            ))}
          </div>

          {/* 3-DAY SCHEDULE SNEAK PEEK */}
          <div className="bg-slate-900/90 border border-slate-800 rounded-3xl p-6 shadow-xl">
            <div className="flex flex-wrap items-center justify-between gap-2 mb-6">
              <div>
                <h2 className="text-xl font-black text-white">📅 Next 3 Days Schedule Sneak Peek</h2>
                <p className="text-xs text-slate-400">
                  Upcoming NBA games with player ownership tags & internal collision alerts
                </p>
              </div>
              <span className="px-3 py-1 rounded-full bg-amber-500/15 text-amber-300 border border-amber-500/30 text-xs font-semibold">
                ⚡ Internal Clash = Both teams owned by same player (+1 Pt guaranteed, -1 Max Possible)
              </span>
            </div>

            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
              {upcomingGames.map((day) => (
                <div key={day.dateCode} className="bg-slate-800/40 border border-slate-800 rounded-2xl p-4">
                  <div className="flex items-center justify-between pb-3 mb-3 border-b border-slate-700/70">
                    <span className="font-black text-sm text-white uppercase tracking-wider">{day.dateLabel}</span>
                    <span className="text-xs text-slate-400 font-semibold">{day.games.length} Games</span>
                  </div>

                  {day.games.length === 0 ? (
                    <div className="text-center py-8 text-xs text-slate-500">
                      No NBA games scheduled on this date.
                    </div>
                  ) : (
                    <div className="space-y-3 max-h-[420px] overflow-y-auto pr-1 custom-scrollbar">
                      {day.games.map((game) => {
                        const awayOwner = teamOwnerMap[game.awayTeam.id];
                        const homeOwner = teamOwnerMap[game.homeTeam.id];
                        const isInternalClash = awayOwner && homeOwner && awayOwner.id === homeOwner.id;

                        return (
                          <div
                            key={game.id}
                            className={`p-3 rounded-xl border transition ${
                              isInternalClash
                                ? 'bg-amber-500/10 border-amber-500/40'
                                : 'bg-slate-900/80 border-slate-800'
                            }`}
                          >
                            <div className="flex items-center justify-between text-[11px] text-slate-400 mb-2">
                              <span className="font-semibold">{game.status}</span>
                              {isInternalClash && (
                                <span className="text-[10px] font-bold text-amber-300 bg-amber-500/20 px-2 py-0.5 rounded-full">
                                  ⚡ {awayOwner.name} Internal Clash
                                </span>
                              )}
                            </div>

                            {/* Away Team */}
                            <div className="flex items-center justify-between py-1">
                              <div className="flex items-center gap-2">
                                <img src={getLogoUrl(game.awayTeam.abbr)} className="w-5 h-5 object-contain" />
                                <span className="font-bold text-sm text-white">{game.awayTeam.abbr}</span>
                                {awayOwner && (
                                  <span className={`text-[10px] font-bold px-2 py-0.5 rounded-md bg-gradient-to-r ${awayOwner.colorClass} text-white`}>
                                    {awayOwner.name}
                                  </span>
                                )}
                              </div>
                              <span className="font-mono font-bold text-sm text-slate-200">
                                {game.awayScore ?? ''}
                              </span>
                            </div>

                            {/* Home Team */}
                            <div className="flex items-center justify-between py-1 mt-1">
                              <div className="flex items-center gap-2">
                                <img src={getLogoUrl(game.homeTeam.abbr)} className="w-5 h-5 object-contain" />
                                <span className="font-bold text-sm text-white">{game.homeTeam.abbr}</span>
                                {homeOwner && (
                                  <span className={`text-[10px] font-bold px-2 py-0.5 rounded-md bg-gradient-to-r ${homeOwner.colorClass} text-white`}>
                                    {homeOwner.name}
                                  </span>
                                )}
                              </div>
                              <span className="font-mono font-bold text-sm text-slate-200">
                                {game.homeScore ?? ''}
                              </span>
                            </div>
                          </div>
                        );
                      })}
                    </div>
                  )}
                </div>
              ))}
            </div>
          </div>

          {/* FULL SNAKE DRAFT RECAP MODAL */}
          {showDraftRecap && (
            <div className="fixed inset-0 bg-black/75 backdrop-blur-sm flex items-center justify-center p-4 z-50">
              <div className="bg-slate-900 border border-slate-700 rounded-3xl max-w-2xl w-full p-6 shadow-2xl max-h-[85vh] flex flex-col">
                <div className="flex items-center justify-between mb-4 pb-3 border-b border-slate-800">
                  <h3 className="text-lg font-black text-white">Snake Draft Pick History (Picks 1–30)</h3>
                  <button onClick={() => setShowDraftRecap(false)} className="text-slate-400 hover:text-white font-bold">✕</button>
                </div>
                <div className="grid grid-cols-2 sm:grid-cols-3 gap-2.5 overflow-y-auto pr-1 custom-scrollbar">
                  {league.picks.map((pick) => {
                    const player = league.players[pick.playerIndex];
                    const team = NBA_TEAMS.find((t) => t.id === pick.teamId);
                    return (
                      <div key={pick.pickNumber} className="p-3 rounded-xl bg-slate-800/70 border border-slate-700/70 flex items-center justify-between">
                        <div>
                          <div className="text-[10px] font-bold text-slate-400">
                            Pick #{pick.pickNumber} • R{Math.floor((pick.pickNumber - 1) / league.numPlayers) + 1}
                          </div>
                          <div className="text-xs font-bold text-white">{player?.name}</div>
                          <div className="text-xs font-extrabold text-orange-400 mt-0.5">{team?.name}</div>
                        </div>
                        {team && <img src={getLogoUrl(team.abbr)} className="w-8 h-8 object-contain" />}
                      </div>
                    );
                  })}
                </div>
              </div>
            </div>
          )}
        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
